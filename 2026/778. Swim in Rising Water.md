# 778. Swim in Rising Water

Use binary search to find the min T to swim from [0, 0] to [n - 1, n - 1].

For every binary search loop:
* Check if the T is possible ==> only use elements <= T to see if we can reach the destination:
*  Use DFS to search from [0, 0], in 4 directions.

* Time: O(n^2 * logn)  # Binary search -- O(logn), DFS -- O(n^2)
* Space: O(n^2)

```python3
class Solution:
    def swimInWater(self, grid: list[list[int]]) -> int:
        n = len(grid)

        def possible(level):
            if level < grid[n - 1][n - 1] or level < grid[0][0]:
                return False

            seen = set()
            def dfs(x, y):
                print(x, y, level)
                if x == n - 1 and y == n - 1:
                    return True

                seen.add((x, y))
                for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
                    r = x + dr
                    c = y + dc
                    if 0 <= r < n and 0 <= c < n and (r, c) not in seen:
                        if grid[r][c] <= level:
                            if dfs(r, c):
                                return True
                
                return False
            return dfs(0, 0)

        def binarysearch(num):
            l = 0
            r = num
            while l < r:
                mid = (l + r) // 2
                if possible(mid):
                    r = mid
                else:
                    l = mid + 1
            return l

        return binarysearch(n * n - 1)
```
